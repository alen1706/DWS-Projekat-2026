CREATE DATABASE IF NOT EXISTS filmoteka_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE filmoteka_db;

CREATE TABLE IF NOT EXISTS korisnici (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ime VARCHAR(50) NOT NULL,
    prezime VARCHAR(50) NOT NULL,
    datum_rodenja DATE NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role ENUM('admin', 'guest') DEFAULT 'guest'
);

CREATE TABLE IF NOT EXISTS zbirke (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    naziv VARCHAR(255) NOT NULL,
    tip ENUM('javna', 'privatna') DEFAULT 'privatna',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS zbirka_stavke (
    id INT AUTO_INCREMENT PRIMARY KEY,
    zbirka_id INT NOT NULL,
    tmdb_id INT NOT NULL,
    media_type ENUM('movie', 'tv') NOT NULL,
    dodano_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (zbirka_id) REFERENCES zbirke(id) ON DELETE CASCADE
);
