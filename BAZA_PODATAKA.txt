CREATE DATABASE IF NOT EXISTS filmoteka_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE filmoteka_db;

CREATE TABLE IF NOT EXISTS korisnici (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ime VARCHAR(50) NOT NULL,
    prezime VARCHAR(50) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role ENUM('admin', 'guest') DEFAULT 'guest',
    bio VARCHAR(100) DEFAULT '',
    profilna_slika VARCHAR(255) DEFAULT 'guest.png',
    status ENUM('active', 'banned') DEFAULT 'active'
);

CREATE TABLE IF NOT EXISTS feedback (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ime VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    poruka TEXT NOT NULL,
    datum_slanja TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS zbirke (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    naziv VARCHAR(255) NOT NULL,
    tip ENUM('javna', 'privatna') DEFAULT 'privatna',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES korisnici(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS zbirka_stavke (
    id INT AUTO_INCREMENT PRIMARY KEY,
    zbirka_id INT NOT NULL,
    tmdb_id INT NOT NULL,
    media_type ENUM('movie', 'tv') NOT NULL,
    dodano_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (zbirka_id) REFERENCES zbirke(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS banovani_korisnici (
    id INT AUTO_INCREMENT PRIMARY KEY,
    korisnik_id INT NOT NULL,
    razlog TEXT NOT NULL,
    datum_bana TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (korisnik_id) REFERENCES korisnici(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS postavke_sistema (
    kljuc VARCHAR(50) PRIMARY KEY,
    vrijednost TEXT
);
